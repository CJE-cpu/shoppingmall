5. Firestore 보안 규칙
- 회원가입된 사용자만 자신의 uid와 같은 users 문서를 생성할 수 있다.
- 생성할 수 있는 필드는 email, nickname, role, createAt으로 제한한다.
- email은 인증 토큰의 이메일과 같아야 한다.
- role은 반드시 user여야 한다.
- createAt은 서버 요청 시간과 같아야 한다.
- 사용자는 자신의 users 문서만 읽을 수 있다.
- 사용자는 자신의 nickname만 수정할 수 있다.
- 사용자가 role, email, createAt을 수정하거나 users 문서를 삭제하는 것은 차단한다.
- users 이외의 모든 Firestore 문서는 현재 기본 차단한다.

테스트 가능한 규칙 코드:
rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    match /users/{userId} {
      allow create: if request.auth != null
        && request.auth.uid == userId
        && request.resource.data.keys().hasOnly([
          'email',
          'nickname',
          'role',
          'createAt'
        ])
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.nickname is string
        && request.resource.data.nickname.size() >= 2
        && request.resource.data.nickname.size() <= 20
        && request.resource.data.role == 'user'
        && request.resource.data.createAt == request.time;

      allow read: if request.auth != null
        && request.auth.uid == userId;

      allow update: if request.auth != null
        && request.auth.uid == userId
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'nickname'
        ])
        && request.resource.data.nickname is string
        && request.resource.data.nickname.size() >= 2
        && request.resource.data.nickname.size() <= 20
        && request.resource.data.role == resource.data.role;

      allow delete: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}

보안 규칙 테스트 항목:
- 인증 사용자가 users/{본인 uid}에 role: user로 문서 생성: 허용
- 비인증 사용자의 users 문서 생성: 거부
- 인증 사용자가 다른 uid의 users 문서 생성 또는 읽기: 거부
- 사용자가 생성 시 role: admin 지정: 거부
- 사용자가 생성 후 role 변경: 거부
- 사용자가 자신의 nickname만 정상 범위에서 변경: 허용
- 사용자가 자신의 email 또는 createAt 변경: 거부
- 사용자가 자신의 문서 삭제: 거부

function isValidProduct(product) {
  return product.keys().hasAll([
      'name',
      'brand',
      'category',
      'categoryValue',
      'price',
      'discountRate',
      'stock',
      'image',
      'images',
      'description',
      'isRecommended',
      'createAt',
      'updateAt'
    ])
    && product.keys().hasOnly([
      'name',
      'brand',
      'category',
      'categoryValue',
      'price',
      'discountRate',
      'stock',
      'image',
      'images',
      'description',
      'isRecommended',
      'createAt',
      'updateAt',
      'legacyId',
      'sortOrder',
      'badge',
      'rating',
      'reviews'
    ])
    && product.name is string
    && product.brand is string
    && product.category is string
    && product.categoryValue is string
    && product.price is number
    && product.price >= 0
    && product.discountRate is number
    && product.discountRate >= 0
    && product.discountRate <= 100
    && product.stock is number
    && product.stock >= 0
    && product.image is string
    && product.images is list
    && product.description is string
    && product.isRecommended is bool
    && (!('legacyId' in product) || product.legacyId is int)
    && (!('sortOrder' in product) || product.sortOrder is int)
    && (!('badge' in product) || product.badge is string)
    && (!('rating' in product) || product.rating is number)
    && (!('reviews' in product) || product.reviews is int);
}

match /products/{productId} {
  allow read: if true;

  allow create: if isAdmin()
    && isValidProduct(request.resource.data)
    && request.resource.data.createAt == request.time
    && request.resource.data.updateAt == request.time;

  allow update: if isAdmin()
    && isValidProduct(request.resource.data)
    && request.resource.data.createAt == resource.data.createAt
    && request.resource.data.updateAt == request.time;

  allow delete: if isAdmin();
}

match /products/{productId} {
  allow read: if true;
  allow create: if isAdmin()
    && isValidProduct(request.resource.data)
    && request.resource.data.createAt == request.time
    && request.resource.data.updateAt == request.time;
  allow update: if (
      isAdmin()
      && isValidProduct(request.resource.data)
      && request.resource.data.createAt == resource.data.createAt
      && request.resource.data.updateAt == request.time
    )
    || (
      request.auth != null
      && isValidProduct(request.resource.data)
      && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
        'stock',
        'updateAt'
      ])
      && request.resource.data.createAt == resource.data.createAt
      && request.resource.data.stock < resource.data.stock
      && request.resource.data.stock >= 0
      && request.resource.data.updateAt == request.time
    );
  allow delete: if isAdmin();
}

match /users/{userId} {
  allow create: if request.auth != null
    && request.auth.uid == userId
    && request.resource.data.keys().hasOnly([
      'email',
      'nickname',
      'role',
      'createAt'
    ])
    && request.resource.data.email == request.auth.token.email
    && request.resource.data.nickname is string
    && request.resource.data.nickname.size() >= 2
    && request.resource.data.nickname.size() <= 20
    && request.resource.data.role == 'user'
    && request.resource.data.createAt == request.time;

  allow get: if request.auth != null
    && (request.auth.uid == userId || isAdmin());

  allow list: if isAdmin();

  allow update: if request.auth != null
    && request.auth.uid == userId
    && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
      'nickname'
    ])
    && request.resource.data.nickname is string
    && request.resource.data.nickname.size() >= 2
    && request.resource.data.nickname.size() <= 20
    && request.resource.data.role == resource.data.role;

  allow delete: if false;
}
